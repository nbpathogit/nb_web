  • Role 2 (คนช่วยตัดนื้อ(คนจด) / Cross Section Assistent): ajax_job2_finish/job2_finish.php → Patient::getAllJob2FinishPhp($conn, $id, $start, $end)

// SQL pattern in Patient::getAllJob1/2/3FinishPhp (ONLY_FULL_GROUP_BY compliant):
SELECT 
    p.id as pid,
    p.pnum as p_pnum, 
    p.phospital_num as p_hn, 
    DATE(p.date_1000) as p_accept,
    CONCAT(p.ppre_name, p.pname, ' ', p.plastname) AS p_patient,
    MAX(j.id) as j_id, 
    MAX(j.patient_id) as j_patient_id, 
    MAX(j.user_id) as j_user_id, 
    MAX(j.job_role_id) as j_job_role_id, 
    MAX(j.jobname) as j_jobname,
    MAX(j.qty) as j_qty,
    p.job1qty as p_job1qty,
    GROUP_CONCAT(CONCAT(j.pre_name, j.name, ' ', j.lastname) SEPARATOR ',') AS j_owners,
    MAX(DATE(j.finish_date)) as j_finish_date
FROM patient as p
LEFT JOIN (select * from job where job.job_role_id=1) as j ON j.patient_id = p.id
WHERE p.date_1000 > '{$start}' AND p.movetotrash = 0
GROUP BY p.id
ORDER BY p.id DESC
